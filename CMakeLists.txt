cmake_minimum_required(VERSION 3.20)
project(dkc_usa_rev1_bringup LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(SNESRECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/upstream/snesrecomp")
if(NOT EXISTS "${SNESRECOMP_ROOT}/runner/runner.cmake")
    message(FATAL_ERROR "Initialize the pinned upstream checkout first")
endif()
include("${SNESRECOMP_ROOT}/runner/runner.cmake")
set(GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/experimental/usa-rev1/generated")
if(NOT EXISTS "${GENERATED_DIR}/source_identity.json")
    message(FATAL_ERROR "Run scripts/generate_rev1.py with the exact USA Rev 1 ROM first")
endif()
file(READ "${GENERATED_DIR}/source_identity.json" GENERATED_IDENTITY)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config/roms/usa-rev1.json" ROM_PROFILE)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/config/upstream.lock.json" SOURCE_LOCK)
string(JSON GENERATED_HASH GET "${GENERATED_IDENTITY}" rom_sha256)
string(JSON EXPECTED_HASH GET "${ROM_PROFILE}" sha256)
if(NOT GENERATED_HASH STREQUAL EXPECTED_HASH)
    message(FATAL_ERROR "Generated sources do not match the USA Rev 1 profile")
endif()
foreach(KEY IN ITEMS commit engine_commit)
    if(KEY STREQUAL "commit")
        set(IDENTITY_KEY upstream_commit)
    else()
        set(IDENTITY_KEY engine_commit)
    endif()
    string(JSON EXPECTED_REV GET "${SOURCE_LOCK}" ${KEY})
    string(JSON GENERATED_REV GET "${GENERATED_IDENTITY}" ${IDENTITY_KEY})
    if(NOT EXPECTED_REV STREQUAL GENERATED_REV)
        message(FATAL_ERROR "Generated sources use a different ${KEY}; regenerate first")
    endif()
endforeach()
set(ENGINE_SOURCES ${SNESRECOMP_RUNNER_SOURCES})
list(REMOVE_ITEM ENGINE_SOURCES
    "${SNESRECOMP_ROOT}/runner/src/keybinds.c"
    "${SNESRECOMP_ROOT}/runner/src/host_report.c")
file(GLOB GENERATED_SOURCES CONFIGURE_DEPENDS
    "${GENERATED_DIR}/*.c")
if(NOT GENERATED_SOURCES)
    message(FATAL_ERROR "Generate the private USA Rev 1 sources first")
endif()
set(REV1_SOURCES ${ENGINE_SOURCES} ${GENERATED_SOURCES}
    upstream/runner/headless_host.c upstream/runner/input_playback.c runner/usa_rev1.c)
add_executable(dkc_rev1_headless ${REV1_SOURCES})
if(WIN32)
    add_executable(dkc_rev1_desktop WIN32 ${REV1_SOURCES} runner/desktop_win32.c)
    target_compile_definitions(dkc_rev1_desktop PRIVATE DKC_REV1_DESKTOP=1 UNICODE _UNICODE)
    target_link_libraries(dkc_rev1_desktop PRIVATE gdi32 winmm comdlg32)
    set_target_properties(dkc_rev1_desktop PROPERTIES OUTPUT_NAME "DKC-Rev1-Teste")
endif()
foreach(TARGET IN ITEMS dkc_rev1_headless dkc_rev1_desktop)
    if(NOT TARGET ${TARGET})
        continue()
    endif()
    target_include_directories(${TARGET} PRIVATE
        recomp/usa-rev1 upstream/runner ${SNESRECOMP_RUNNER_INCLUDE_DIRS})
    target_compile_definitions(${TARGET} PRIVATE
        SNESRECOMP_TRACE=0 SNESRECOMP_REVERSE_DEBUG=0
        SNESRECOMP_EXTERNAL_RAM_ROUTINE_GUARDS=1 SYSTEM_VOLUME_MIXER_AVAILABLE=0)
    if(MSVC)
        target_compile_options(${TARGET} PRIVATE /W3 /utf-8)
        target_compile_definitions(${TARGET} PRIVATE _CRT_SECURE_NO_WARNINGS)
        target_link_libraries(${TARGET} PRIVATE ws2_32 user32 advapi32)
    endif()
    target_link_libraries(${TARGET} PRIVATE ${SNESRECOMP_RUNNER_LIBRARIES})
endforeach()
